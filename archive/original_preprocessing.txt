message("\nPASSO 4: Preparando matriz com Gene Symbols para deconvolution...")

  filter(ensembl_gene_id %in% rownames(counts_autossomais)) %>%
  filter(!is.na(external_gene_name) & external_gene_name != "") %>%
  distinct(ensembl_gene_id, .keep_all = TRUE)

  rownames_to_column("ensembl_gene_id")

# Juntar contagens com os símbolos de genes (usando R base para selecionar colunas)
counts_with_symbols <- inner_join(
  counts_autossomais_df,
  # Usando colchetes para selecionar as colunas, o que evita o conflito de nomes
  gene_map[, c("ensembl_gene_id", "external_gene_name")],
  by = "ensembl_gene_id"
)

# Agrupar por Gene Symbol e somar as contagens.
counts_aggregated_by_symbol <- counts_with_symbols %>%
  # CORREÇÃO: Especificar dplyr::select para remover a coluna
  dplyr::select(-ensembl_gene_id) %>%
  group_by(external_gene_name) %>%
  summarise(across(where(is.numeric), sum), .groups = 'drop')

# Converter de volta para uma matriz com Gene Symbols como rownames
counts_autossomais_gene_symbol <- counts_aggregated_by_symbol %>%
  column_to_rownames("external_gene_name")

# PASSO 6: ANÁLISE E VISUALIZAÇÃO DOS RESULTADOS DA DECONVOLUTION
#-------------------------------------------------------------------------------
message("\nPASSO 6: Gerando gráfico das proporções celulares...")

# Adicionar a coluna 'Condition' ao data frame de frações para plotagem
fractions_df <- as.data.frame(fractions) %>%
  rownames_to_column("sample")

col_data_df <- as.data.frame(col_data_filtered) %>%
  rownames_to_column("sample")

fractions_with_condition <- left_join(fractions_df, 
                                      col_data_df %>% dplyr::select(sample, Condition), 
                                      by = "sample")

# Transformar dados para o formato longo (long format)
plot_data <- fractions_with_condition %>%
  pivot_longer(
    cols = -c(sample, Condition),
    names_to = "CellType",
    values_to = "Proportion"
  )

# --- CORREÇÃO 1: Definir a ordem das condições ---
# Transformamos 'Condition' em um fator para garantir que 'pre' venha antes de 'pos'
plot_data$Condition <- factor(plot_data$Condition, levels = c("pre", "pos"))

# Calcula a média por grupo para as barras do gráfico
summary_data <- plot_data %>%
  group_by(Condition, CellType) %>%
  summarise(mean_prop = mean(Proportion, na.rm = TRUE), .groups = "drop")

# Teste estatístico (Wilcoxon) para cada tipo celular
wilcox_results <- plot_data %>%
  group_by(CellType) %>%
  summarise(
    p_value = wilcox.test(Proportion ~ Condition)$p.value,
    .groups = "drop"
  ) %>%
  mutate(
    p_adj = p.adjust(p_value, method = "BH"),
    signif = case_when(
      p_adj <= 0.001 ~ "***",
      p_adj <= 0.01  ~ "**",
      p_adj <= 0.05  ~ "*",
      TRUE           ~ ""
    )
  )

# Preparar posições para os asteriscos de significância
text_positions <- plot_data %>%
  group_by(CellType) %>%
  summarise(max_prop = max(Proportion, na.rm = TRUE)) %>%
  left_join(wilcox_results, by = "CellType") %>%
  mutate(y_pos = max_prop * 1.05) # Posição 5% acima do ponto máximo

# --- AJUSTES NO GRÁFICO ---
# Definimos as cores manualmente para um controle total
cor_pre <- "#4682B4"  # Um azul mais forte (SteelBlue)
cor_pos <- "#B22222"  # Um vermelho mais forte (Firebrick)

final_plot <- ggplot(summary_data, aes(x = CellType, y = mean_prop, fill = Condition)) +
  # Barras representando a média, com a transparência desejada
  geom_bar(stat = "identity", position = position_dodge(0.9), alpha = 0.7) +
  
  # Pontos para cada amostra individual
  geom_jitter(
    data = plot_data,
    # Mapeamos a cor dos pontos para a Condição
    aes(y = Proportion, color = Condition), 
    position = position_jitterdodge(jitter.width = 0.2, dodge.width = 0.9),
    size = 2,
    # Pontos menos transparentes para se destacarem mais
    alpha = 0.9
  ) +
  
  # Asteriscos de significância
  geom_text(
    data = text_positions %>% filter(signif != ""),
    aes(x = CellType, y = y_pos, label = signif),
    inherit.aes = FALSE,
    size = 6
  ) +
  
  # --- NOVAS LINHAS: Controle manual das cores ---
  # Define a cor de PREENCHIMENTO (fill) para as barras
  scale_fill_manual(values = c("pre" = cor_pre, "pos" = cor_pos)) +
  # Define a cor do CONTORNO/PONTOS (color) para os pontos
  scale_color_manual(values = c("pre" = cor_pre, "pos" = cor_pos)) +
  
  labs(
    title = "Proporção de Tipos Celulares por Condição",
    x = "Tipo Celular",
    y = "Proporção Estimada"
  ) +
  # Remove as legendas duplicadas (uma para fill e outra para color)
  guides(color = "none", fill = guide_legend(title = "Condição")) +
  theme_minimal(base_size = 14) +
  theme(
    axis.text.x = element_text(angle = 45, hjust = 1),
    legend.position = "top",
    panel.grid.major.x = element_blank()
  )

print(final_plot)

# 3. CRIAR UMA FUNÇÃO DE PLOTAGEM REUTILIZÁVEL (Mesma de antes)
#-------------------------------------------------------------------------------

plot_correlation <- function(data, gene_x, gene_y, method = "pearson") {
  if (!all(c(gene_x, gene_y) %in% names(data))) {
    warning(paste("Um ou ambos os genes", gene_x, "ou", gene_y, "não foram encontrados."))
    return(NULL)
  }
  
  corr_test <- cor.test(data[[gene_x]], data[[gene_y]], method = method)
  r_value <- corr_test$estimate
  p_value <- corr_test$p.value
  
  subtitle_text <- sprintf("Correlação de Pearson: r = %.2f, p = %.3f", r_value, p_value)
  
  p <- ggplot(data, aes(x = .data[[gene_x]], y = .data[[gene_y]])) +
    geom_point(aes(color = col_data_filtered$Condition), size = 3, alpha = 0.8) +
    geom_smooth(method = "lm", se = FALSE, color = "gray20", linetype = "dashed") +
    scale_color_manual(values = c("pre" = "#4682B4", "pos" = "#B22222"), name = "Condição") +
    labs(
      title = paste("Correlação entre", gene_x, "e", gene_y),
      subtitle = subtitle_text,
      x = paste("Expressão Log2(n+1) de", gene_x),
      y = paste("Expressão Log2(n+1) de", gene_y)
    ) +
    theme_bw() +
    theme(legend.position = "top")
  
  return(p)
}
