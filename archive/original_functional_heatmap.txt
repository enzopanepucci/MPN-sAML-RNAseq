# --- 1. CARREGAR BIBLIOTECAS ---
library(ggplot2)
library(dplyr)
library(stringr)

# --- 2. PROCESSAR E CATEGORIZAR OS GENES ---
# (Este bloco permanece o mesmo)
genes_categorizados <- tabela_final_genes %>%
  mutate(Function = str_remove(Function, " \\[Source.*")) %>%
  mutate(
    FunctionalGroup = case_when(
      grepl("RNA|ribosomal|splicing|nucleolar", Function, ignore.case = TRUE) ~ "RNA Machinery",
      grepl("apoptosis|caspase|BCL|BIRC3|death", Function, ignore.case = TRUE) ~ "Apoptosis Regulation",
      grepl("chemokine|CCL|CXCL|CCR|CX3CR", GeneSymbol, ignore.case = TRUE) ~ "Chemokines & Receptors",
      grepl("interleukin|IFN|TNF|IDO1|TGF", GeneSymbol, ignore.case = TRUE) ~ "Cytokines & Signaling",
      grepl("integrin|ITG|adhesion|SELP|ICAM1", GeneSymbol, ignore.case = TRUE) ~ "Adhesion & Niche",
      grepl("toll like|TLR|NOD2|NLRP|inflammasome", GeneSymbol, ignore.case = TRUE) ~ "Innate Sensing",
      TRUE ~ "Other"
    )
  )

# --- 3. SELECIONAR OS GENES MAIS IMPORTANTES (ATÉ 4 POR GRUPO) ---
# (Este bloco permanece o mesmo)
genes_para_plotar <- genes_categorizados %>%
  filter(FunctionalGroup != "Other") %>%
  group_by(FunctionalGroup) %>%
  slice_max(order_by = abs(log2FC), n = 10) %>%
  ungroup()

# --- 4. DEFINIR COORDENADAS E ADICIONAR O TIPO DE MODULAÇÃO ---
posicoes_grupos <- data.frame(
  FunctionalGroup = c("RNA Machinery", "Apoptosis Regulation", "Chemokines & Receptors", "Innate Sensing", "Adhesion & Niche", "Cytokines & Signaling"),
  x = c(2.5, 2.5, 7.5, 7.5, 7.5, 7.5),
  y = c(8, 3, 9, 6.5, 4, 1.5),
  # *** NOVA COLUNA PARA DEFINIR A COR DA CAIXA ***
  type = c("UP", "DOWN", "DOWN", "DOWN", "DOWN", "DOWN") 
)

# (Este bloco permanece o mesmo)
dados_finais_figura <- genes_para_plotar %>%
  inner_join(posicoes_grupos, by = "FunctionalGroup") %>%
  group_by(FunctionalGroup) %>%
  mutate(
    angle = seq(0, 2 * pi, length.out = n() + 1)[1:n()],
    x_end = x + 1.8 * cos(angle),
    y_end = y + 1.0 * sin(angle)
  ) %>%
  ungroup() %>%
  mutate(
    label = paste0(GeneSymbol, " (", round(log2FC, 2), ")"),
    type = ifelse(log2FC > 0, "UP", "DOWN")
  )

# --- 5. CONSTRUIR A FIGURA COM CAIXAS COLORIDAS ---

# Cores
up_color <- "#d95f02"       # Laranja Sólido
down_color <- "#1f78b4"     # Azul Sólido
light_up_color <- "#fee6ce"  # Laranja Claro
light_down_color <- "#deebf7" # Azul Claro

figura_final_colorida <- ggplot() +
  # Limites e tema
  coord_cartesian(xlim = c(0, 10), ylim = c(0, 11.5)) +
  theme_void() +
  
  # Linha divisória e Títulos (sem alteração)
  geom_vline(xintercept = 5, linetype = "dashed", color = "grey60", linewidth = 0.5) +
  annotate("text", x = 2.5, y = 11, label = "Axis I: Internal Reprogramming", fontface = "bold", size = 6) +
  annotate("text", x = 7.5, y = 11, label = "Axis II: External Disconnection", fontface = "bold", size = 6) +
  
  # Ramificações (desenhadas primeiro para ficarem no fundo)
  geom_segment(data = dados_finais_figura,
               aes(x = x, y = y, xend = x_end, yend = y_end, color = type),
               linewidth = 0.6) +
  
  # *** ALTERAÇÃO PRINCIPAL: CAIXAS DE GRUPO COLORIDAS ***
  # Desenhar as caixas UP com fundo laranja claro
  geom_label(data = filter(posicoes_grupos, type == "UP"), 
             aes(x = x, y = y, label = str_wrap(FunctionalGroup, width = 15)),
             fontface = "bold", size = 3.5, fill = light_up_color, color = "black", 
             label.padding = unit(0.3, "lines")) +
  
  # Desenhar as caixas DOWN com fundo azul claro
  geom_label(data = filter(posicoes_grupos, type == "DOWN"), 
             aes(x = x, y = y, label = str_wrap(FunctionalGroup, width = 15)),
             fontface = "bold", size = 3.5, fill = light_down_color, color = "black", 
             label.padding = unit(0.3, "lines")) +
  
  # Labels dos Genes (com cores sólidas)
  geom_label(data = dados_finais_figura,
             aes(x = x_end, y = y_end, label = label, fill = type),
             color = "white", size = 3, fontface = "bold") +
  
  # Escalas e Título geral
  scale_fill_manual(values = c("UP" = up_color, "DOWN" = down_color)) +
  scale_color_manual(values = c("UP" = up_color, "DOWN" = down_color)) +
  theme(legend.position = "none",
        plot.title = element_text(hjust = 0.5, size = 18, face = "bold", margin = margin(b = 10))) +
  labs(title = "Figure 4: A Dual-Axis Model for Leukemic Progression")

# Visualizar a figura
print(figura_final_colorida)

# Salvar em alta resolução
ggsave("Figura_4_Eixos_Final_Colorida.png", figura_final_colorida, width = 15, height = 9, dpi = 300)

#-------------------------------------------------------------------------------
# 1. PREPARAR A LISTA HÍBRIDA DE GENES (LÓGICA MANTIDA)
#-------------------------------------------------------------------------------
degs_symbols <- rownames(scaled_matrix_sorted)

genes_for_hybrid_heatmap_df <- all_genes_categorized %>%
  filter(!is.na(FunctionalGroup)) %>%
  filter(FunctionalGroup != "RNA Machinery" | (GeneSymbol %in% degs_symbols))

genes_to_plot_hybrid <- genes_for_hybrid_heatmap_df$GeneSymbol

#-------------------------------------------------------------------------------
# 2. PREPARAR A MATRIZ E OS DADOS PARA ANOTAÇÃO
#-------------------------------------------------------------------------------

# Pipeline de preparação da matriz (mesma de antes)
counts_deseq_normalized <- counts(dds_noXY, normalized = TRUE)
counts_df_symbol <- as.data.frame(counts_deseq_normalized) %>%
  rownames_to_column("ensembl_gene_id") %>%
  left_join(gene_map %>% dplyr::select(ensembl_gene_id, external_gene_name), by = "ensembl_gene_id") %>%
  filter(!is.na(external_gene_name) & external_gene_name != "") %>%
  distinct(external_gene_name, .keep_all = TRUE) %>%
  dplyr::select(-ensembl_gene_id) %>%
  column_to_rownames("external_gene_name")
matrix_to_scale <- counts_df_symbol[rownames(counts_df_symbol) %in% genes_to_plot_hybrid, ]
scaled_matrix_full <- t(scale(t(as.matrix(matrix_to_scale))))
scaled_matrix_full <- scaled_matrix_full[complete.cases(scaled_matrix_full), ]
sample_order <- rownames(col_annotation_df_sorted)
scaled_matrix_hybrid_sorted <- scaled_matrix_full[, sample_order]

# Preparar o data frame para o split de linhas
row_split_df_hybrid <- genes_for_hybrid_heatmap_df %>%
  as.data.frame() %>%
  dplyr::select(GeneSymbol, FunctionalGroup) %>%
  filter(GeneSymbol %in% rownames(scaled_matrix_hybrid_sorted)) %>%
  { .[match(rownames(scaled_matrix_hybrid_sorted), .$GeneSymbol), ] }
group_order <- c("Chemokines & Receptors", "Cytokines & Signaling", "Adhesion & Niche",
                 "Innate Immunity Effectors", "Apoptosis Regulation", "RNA Machinery")
row_split_df_hybrid$FunctionalGroup <- factor(row_split_df_hybrid$FunctionalGroup, levels = group_order)

#-------------------------------------------------------------------------------
# 3. SOLUÇÃO: CRIAR RÓTULOS SELETIVOS COM 'anno_mark' (SINTAXE CORRIGIDA)
#-------------------------------------------------------------------------------

# Adicionar o padj ao nosso data frame de split para poder ranquear os genes
degs_padj <- significant_degs %>% 
  as.data.frame() %>% 
  # --- CORREÇÃO APLICADA AQUI ---
  # Especificar dplyr::select para resolver o conflito de pacotes
  dplyr::select(padj) %>% 
  rownames_to_column("EnsemblID")

degs_padj_symbol <- left_join(
  degs_padj, 
  # Especificar dplyr::select aqui também, por segurança
  gene_map %>% dplyr::select(ensembl_gene_id, external_gene_name), 
  by=c("EnsemblID"="ensembl_gene_id")
) %>%
  dplyr::select(external_gene_name, padj) %>% 
  dplyr::rename(GeneSymbol = external_gene_name)

row_split_df_hybrid <- left_join(row_split_df_hybrid, degs_padj_symbol, by = "GeneSymbol")

# --- CORREÇÃO APLICADA AQUI: USANDO SPLIT-APPLY-COMBINE ---

# 1. Preparar e dividir o data frame em uma lista, um elemento por FunctionalGroup
df_to_process <- row_split_df_hybrid %>%
  as.data.frame() %>% # Forçar a conversão para data.frame simples
  filter(!is.na(padj))

list_by_group <- split(df_to_process, df_to_process$FunctionalGroup)

# 2. Aplicar a lógica de seleção para cada grupo na lista
list_of_top_genes <- lapply(list_by_group, function(group_df) {
  # Ordenar o data frame do grupo atual por padj
  group_df_sorted <- group_df[order(group_df$padj), ]
  
  # Determinar quantos genes pegar
  n_to_keep <- ifelse(unique(group_df_sorted$FunctionalGroup) == "RNA Machinery", 15, 8)
  
  # Pegar as N primeiras linhas
  top_genes_df <- head(group_df_sorted, n = n_to_keep)
  
  return(top_genes_df)
})

# 3. Combinar a lista de resultados de volta em um único data frame
top_genes_df_final <- bind_rows(list_of_top_genes)

# 4. Extrair os Gene Symbols para rotulagem
genes_para_rotular_seletivamente <- top_genes_df_final$GeneSymbol


# Criar a anotação de linha (este código agora funcionará)
row_anno <- rowAnnotation(
  show_labels = anno_mark(
    at = which(rownames(scaled_matrix_hybrid_sorted) %in% genes_para_rotular_seletivamente), 
    labels = rownames(scaled_matrix_hybrid_sorted)[which(rownames(scaled_matrix_hybrid_sorted) %in% genes_para_rotular_seletivamente)],
    labels_gp = gpar(fontsize = 9)
  )
)
#-------------------------------------------------------------------------------
# 4. GERAR O HEATMAP FINAL COM RÓTULOS OTIMIZADOS
#-------------------------------------------------------------------------------

final_poster_heatmap <- Heatmap(
  scaled_matrix_hybrid_sorted,
  name = "Z-Score",
  col = colorRamp2(c(-2, 0, 2), c("blue", "white", "red")),
  top_annotation = ha,
  column_title = "Functionally Grouped Genes",
  
  # Usar a nova anotação de linha para os rótulos
  right_annotation = row_anno,
  
  row_split = row_split_df_hybrid$FunctionalGroup,
  row_title_rot = 0,
  row_title_gp = gpar(fontsize = 12, fontface = "bold"),
  
  cluster_rows = TRUE,
  cluster_columns = FALSE,
  
  # Esconder os nomes originais das linhas, pois agora usamos anno_mark
  show_row_names = FALSE
)
