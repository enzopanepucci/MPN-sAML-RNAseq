# 2. VOLCANO PLOT

padj_cutoff <- 0.05
lfc_cutoff <- 0.5

volcano_df <- res_noXY_df %>%
  mutate(
    padj_for_plot = ifelse(is.na(padj), NA, pmax(padj, .Machine$double.xmin)),
    neg_log10_padj = -log10(padj_for_plot),
    
    Volcano_category = case_when(
      is.na(padj) ~ "Not tested",
      
      padj < padj_cutoff & log2FoldChange >= lfc_cutoff ~
        "Upregulated in post-leukemic progression",
      
      padj < padj_cutoff & log2FoldChange <= -lfc_cutoff ~
        "Downregulated in post-leukemic progression",
      
      padj < padj_cutoff & abs(log2FoldChange) < lfc_cutoff ~
        "Statistically significant, small effect",
      
      padj >= padj_cutoff & log2FoldChange >= 0 ~
        "Post-side, not significant",
      
      padj >= padj_cutoff & log2FoldChange < 0 ~
        "Pre-side, not significant",
      
      TRUE ~ "Not tested"
    )
  )

volcano_colors <- c(
  "Upregulated in post-leukemic progression" = "#B2182B",
  "Downregulated in post-leukemic progression" = "#2166AC",
  "Statistically significant, small effect" = "#7B3294",
  "Post-side, not significant" = "#F4A3A8",
  "Pre-side, not significant" = "#92C5DE",
  "Not tested" = "#BDBDBD"
)

label_genes <- volcano_df %>%
  filter(
    !is.na(padj),
    padj < padj_cutoff,

# ============================================================
# counts_noXY -> TPM -> log2(TPM/10 + 1)
# Mixture file compatible with TISCH2-derived signature matrix
# ============================================================

library(rtracklayer)
library(GenomicRanges)
library(dplyr)
library(tibble)

# -----------------------------
# 1. Inputs
# -----------------------------

# Matriz raw counts sem genes X/Y
# linhas = genes Ensembl
# colunas = amostras
counts_mat <- as.matrix(counts_noXY)

# Caminho para o mesmo GTF usado no alinhamento/contagem
gtf_file <- "Homo_sapiens.GRCh38.gtf"

# Prefixo dos arquivos de saída
out_prefix <- "mixture_counts_noXY_TISCH2_scale"

# -----------------------------
# 2. Limpar IDs Ensembl
# -----------------------------

rownames(counts_mat) <- sub("\\..*$", "", rownames(counts_mat))

# -----------------------------
# 3. Importar GTF e calcular gene length
# -----------------------------

gtf <- rtracklayer::import(gtf_file)

exons <- gtf[gtf$type == "exon"]

exons$gene_id_clean <- sub("\\..*$", "", exons$gene_id)

exons_by_gene <- split(exons, exons$gene_id_clean)

gene_length_bp <- vapply(
  reduce(exons_by_gene),
  function(x) sum(width(x)),
  numeric(1)
)

gene_annot <- unique(data.frame(
  gene_id = sub("\\..*$", "", exons$gene_id),
  gene_symbol = exons$gene_name,
  stringsAsFactors = FALSE
))

# -----------------------------
# 4. Manter genes em comum
# -----------------------------

common_genes <- intersect(rownames(counts_mat), names(gene_length_bp))

counts_mat <- counts_mat[common_genes, , drop = FALSE]
gene_length_bp <- gene_length_bp[common_genes]

valid_genes <- !is.na(gene_length_bp) & gene_length_bp > 0

counts_mat <- counts_mat[valid_genes, , drop = FALSE]
gene_length_bp <- gene_length_bp[valid_genes]

message("Genes after matching with GTF: ", nrow(counts_mat))

# -----------------------------
# 5. Counts -> TPM
# -----------------------------

gene_length_kb <- gene_length_bp / 1000

rpk <- counts_mat / gene_length_kb[rownames(counts_mat)]

tpm <- sweep(
  rpk,
  2,
  colSums(rpk, na.rm = TRUE),
  "/"
) * 1e6

# Checagem: cada coluna deve somar ~1e6
print(round(colSums(tpm)[1:min(5, ncol(tpm))], 2))

# -----------------------------
# 6. Converter Ensembl -> Gene Symbol
# -----------------------------

annot_use <- gene_annot[match(rownames(tpm), gene_annot$gene_id), ]

tpm_df <- as.data.frame(tpm)
tpm_df$gene_symbol <- annot_use$gene_symbol

tpm_df <- tpm_df %>%
  filter(!is.na(gene_symbol), gene_symbol != "")

# Agregar símbolos duplicados em escala linear TPM
# Usar soma é mais apropriado antes do log
tpm_symbol <- tpm_df %>%
  group_by(gene_symbol) %>%
  summarise(across(where(is.numeric), sum), .groups = "drop")

tpm_symbol <- as.data.frame(tpm_symbol)
rownames(tpm_symbol) <- tpm_symbol$gene_symbol
tpm_symbol$gene_symbol <- NULL

message("Gene symbols after aggregation: ", nrow(tpm_symbol))

# -----------------------------
# 7. TPM -> escala TISCH2
# -----------------------------

mixture_tisch2_scale <- log2(tpm_symbol / 10 + 1)

# -----------------------------
# 8. Salvar objetos
# -----------------------------

saveRDS(
  tpm_symbol,
  file = paste0(out_prefix, "_TPM_geneSymbol.rds")
)

saveRDS(
  mixture_tisch2_scale,
  file = paste0(out_prefix, "_log2TPMdiv10plus1_geneSymbol.rds")
)

# Arquivo .txt para deconvolution
mixture_out <- mixture_tisch2_scale %>%
  as.data.frame() %>%
  rownames_to_column("GeneSymbol")

write.table(
  mixture_out,
  file = paste0(out_prefix, "_log2TPMdiv10plus1_geneSymbol.txt"),
  sep = "\t",
  quote = FALSE,
  row.names = FALSE
)

# Também salvar TPM linear, caso queira rodar análise de sensibilidade
tpm_out <- tpm_symbol %>%
  as.data.frame() %>%
  rownames_to_column("GeneSymbol")

write.table(
  tpm_out,
  file = paste0(out_prefix, "_TPM_geneSymbol.txt"),
  sep = "\t",
  quote = FALSE,
  row.names = FALSE
)

# -----------------------------
# 9. Checagens finais
# -----------------------------

message("Final mixture dimensions:")
print(dim(mixture_tisch2_scale))

message("Summary of TISCH2-scale mixture:")
print(summary(as.vector(as.matrix(mixture_tisch2_scale))))

message("Files saved:")
print(list.files(pattern = out_prefix))

plot_data <- CIBERSORTx_results %>%
  rownames_to_column("Patient_ID") %>%
  pivot_longer(
    cols = -c(Patient_ID, Condition),
    names_to = "CellState",
    values_to = "Proportion"
  )

summary_data <- plot_data %>%
  group_by(Condition, CellState) %>%
  summarise(
    mean_prop = mean(Proportion, na.rm = TRUE),
    .groups = "drop"
  )

# Como Condition tem 4 grupos, usar Kruskal-Wallis
stat_results <- plot_data %>%
  group_by(CellState) %>%
  summarise(
    p_value = kruskal.test(Proportion ~ Condition)$p.value,
    .groups = "drop"
  ) %>%
  mutate(
    p_adj = p.adjust(p_value, method = "BH"),
    signif = case_when(
      p_adj <= 0.001 ~ "***",
      p_adj <= 0.01  ~ "**",
      p_adj <= 0.05  ~ "*",
      TRUE           ~ ""
    )
  )

text_positions <- plot_data %>%
  group_by(CellState) %>%
  summarise(
    max_prop = max(Proportion, na.rm = TRUE),
    .groups = "drop"
  ) %>%
  left_join(stat_results, by = "CellState") %>%
  mutate(y_pos = max_prop + 0.02)

p_prop <- ggplot(summary_data, aes(x = CellState, y = mean_prop, fill = Condition)) +

    colour = "grey30",
    linewidth = 0.2
  ) +
  
  geom_jitter(
    data = plot_data,
    aes(x = CellState, y = Proportion, color = Condition),
    position = position_jitterdodge(jitter.width = 0.2, dodge.width = 0.9),
    size = 1.8,
    shape = 21,
    alpha = 0.85,
    inherit.aes = FALSE
  ) +
  
  geom_text(
    data = text_positions %>% filter(signif != ""),
    aes(x = CellState, y = y_pos, label = signif),
    inherit.aes = FALSE,
    vjust = 0,
    size = 5
  ) +
  
  scale_fill_manual(values = condition_colors, name = "Sampling point") +
  scale_color_manual(values = condition_colors, name = "Sampling point") +
  scale_y_continuous(labels = scales::percent_format(accuracy = 1)) +
  
  labs(
    title = "CIBERSORTx-estimated cell-state proportions across sampling points",
