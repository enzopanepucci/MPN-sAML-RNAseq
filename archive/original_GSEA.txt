# Esta nova função é mais segura e baseia-se diretamente nas listas de vias finais.
prepare_plot_data_robust <- function(res_padrao_full, res_custom_full, pathways_padrao_sig, pathways_custom_sig, db_name) {
  
  # A. Criar o status a partir das listas de vias que SABEMOS que estão corretas
  all_sig_pathways <- union(pathways_padrao_sig, pathways_custom_sig)
  
  status_df <- tibble(pathway = all_sig_pathways) %>%
    mutate(Status = case_when(
      pathway %in% pathways_padrao_sig & pathway %in% pathways_custom_sig ~ "Comum",
      pathway %in% pathways_padrao_sig & !(pathway %in% pathways_custom_sig) ~ "Exclusiva (Padrão)",
      !(pathway %in% pathways_padrao_sig) & pathway %in% pathways_custom_sig ~ "Exclusiva (Custom)"
    ))
  
  # B. Preparar os dados de NES de cada método
  df_padrao <- res_padrao_full %>% as_tibble() %>% dplyr::select(pathway, NES_padrao = NES)
  df_custom <- res_custom_full %>% as_tibble() %>% dplyr::select(pathway, NES_custom = NES)
  
  # C. Juntar tudo
  plot_data <- status_df %>%
    left_join(df_padrao, by = "pathway") %>%
    left_join(df_custom, by = "pathway") %>%
    mutate(
      NES_padrao = ifelse(is.na(NES_padrao), 0, NES_padrao),
      NES_custom = ifelse(is.na(NES_custom), 0, NES_custom)
    ) %>%
    mutate(avg_NES = (NES_padrao + NES_custom) / 2) %>%
    arrange(avg_NES) %>%
    pivot_longer(cols = c("NES_padrao", "NES_custom"), names_to = "Metodo", values_to = "NES") %>%
    filter(NES != 0) %>%
    mutate(Metodo = recode(Metodo, "NES_padrao" = "Métrica Padrão", "NES_custom" = "Sua Métrica Híbrida"))
  
  # D. Preparar informações do eixo Y
  y_axis_info <- plot_data %>% distinct(pathway, Status, avg_NES) %>% arrange(avg_NES)
  y_axis_labels <- y_axis_info$pathway
  y_axis_colors <- case_when(
    y_axis_info$Status == "Comum" ~ "black",
    y_axis_info$Status == "Exclusiva (Padrão)" ~ "blue",
    y_axis_info$Status == "Exclusiva (Custom)" ~ "red",
    TRUE ~ "grey"
  )
  
  plot_data$pathway <- factor(plot_data$pathway, levels = y_axis_labels)
  
  return(list(data = plot_data, colors = y_axis_colors))
}

# Função auxiliar aprimorada para criar os gráficos de barras de NES
plot_gsea_results <- function(fgsea_res, title, top_n = Inf) {
  
  plot_data_full <- fgsea_res %>%
    as_tibble() %>%
    filter(padj < 0.05)
  
  # Aplicar o filtro top_n apenas se ele for um número finito
  if (is.finite(top_n)) {
    plot_data <- plot_data_full %>%
      slice_max(order_by = abs(NES), n = top_n)
    subtitle_text <- paste("Mostrando as Top", top_n, "vias por |NES|")
  } else {
    plot_data <- plot_data_full
    subtitle_text <- paste("Mostrando todas as", nrow(plot_data), "vias significativas")
  }
  
  plot_data <- plot_data %>%
    arrange(NES) %>%
    mutate(pathway = factor(pathway, levels = pathway))
  
  if (nrow(plot_data) == 0) return(NULL)
  
  p <- ggplot(plot_data, aes(x = NES, y = pathway, fill = NES > 0)) +
    geom_col() +
    scale_fill_manual(values = c("TRUE" = "#B22222", "FALSE" = "#4682B4"), guide = "none") +
    labs(title = title, subtitle = subtitle_text, x = "NES", y = NULL) +
    theme_minimal() +
    theme(axis.text.y = element_text(size = 8))
  
  return(list(plot = p, data = plot_data))
}

# 1. FUNÇÃO DE PLOTAGEM
#-------------------------------------------------------------------------------
# Esta função auxiliar permanece a mesma.
plot_gsea_results <- function(fgsea_res_filtered, title, subtitle = NULL) {
  plot_data <- fgsea_res_filtered %>%
    as_tibble() %>%
    arrange(NES) %>%
    mutate(pathway = factor(pathway, levels = pathway))
  
  if (nrow(plot_data) == 0) {
    message(paste("Nenhuma via encontrada para:", title, "-", subtitle))
    return(NULL)
  }
  
  p <- ggplot(plot_data, aes(x = NES, y = pathway, fill = NES > 0)) +
    geom_col() +
    scale_fill_manual(values = c("TRUE" = "#B22222", "FALSE" = "#4682B4"), guide = "none") +
    labs(title = title, subtitle = subtitle, x = "NES", y = NULL) +
    theme_minimal() +
    theme(axis.text.y = element_text(size = 8))
  
  return(list(plot = p, n_pathways = nrow(plot_data)))
}

# Esta nova função recebe o NOME do objeto como uma string.
create_and_save_summary <- function(object_name, output_filename) {
  
  # A verificação agora funciona corretamente: "O objeto com este nome existe no ambiente?"
  if (!exists(object_name, envir = .GlobalEnv)) {
    warning(paste("Objeto", object_name, "não encontrado. Pulando."))
    return()
  }
  
  # Pega o objeto real do ambiente usando seu nome
  fgsea_result_object <- get(object_name, envir = .GlobalEnv)
  
  # Processa a tabela (o resto da lógica é a mesma)
  summary_table <- fgsea_result_object %>%
    as_tibble() %>%
    filter(padj < 0.05) %>%
    dplyr::select(pathway, NES, padj, size) %>%
    arrange(desc(NES))
  
  # Salva o arquivo CSV
  fwrite(summary_table, output_filename)
  
  message(paste("Arquivo salvo:", output_filename, "com", nrow(summary_table), "vias."))
}
